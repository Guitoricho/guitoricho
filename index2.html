// Système de gestion de données simple const AppData = { user: JSON.parse(localStorage.getItem('user')) || { username: "Invité", status: "Online" }, messages: [], save() { localStorage.setItem('user', JSON.stringify(this.user)); }, sendMessage(text) { const msg = { author: this.user.username, content: text, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }; this.renderMessage(msg); }, renderMessage(msg) { const container = document.getElementById('chat-messages'); const html = `

${msg.author} ${msg.time}

${msg.content}

`; container.innerHTML += html; container.scrollTop = container.scrollHeight; } }; // Initialisation document.getElementById('username-display').innerText = AppData.user.username; document.getElementById('chat-input').addEventListener('keypress', (e) => { if(e.key === 'Enter' && e.target.value.trim() !== "") { AppData.sendMessage(e.target.value); e.target.value = ""; } });
